% Translate the renderer's English section names to academic conventions.
\newcommand{\resumesection}[1]{%
  \def\tmptitle{#1}%
  \IfStrEq{#1}{Summary}{\def\tmptitle{Academic \& Research Profile}}{}%
  \IfStrEq{#1}{Experience}{\def\tmptitle{Research \& Professional Experience}}{}%
  \IfStrEq{#1}{Projects}{\def\tmptitle{Research Projects}}{}%
  \IfStrEq{#1}{Education}{\def\tmptitle{Education \& Degrees}}{}%
  \IfStrEq{#1}{Skills}{\def\tmptitle{Research Interests \& Methods}}{}%
  \IfStrEq{#1}{Certifications}{\def\tmptitle{Publications, Awards \& Service}}{}%
  \IfStrEq{#1}{Languages}{\def\tmptitle{Languages}}{}%
  \vspace{1em}%
  {\color{accent}\scshape\bfseries\large \tmptitle}\par
  \vspace{0.15em}{\textcolor{accent}{\rule{\textwidth}{0.5pt}}}\par
  \vspace{0.45em}%
}

\newcommand{\resumeentry}[4]{%
  {\bfseries\color{ink} #1}\hfill{\color{muted}#3}\\[0.05em]
  {\itshape\color{muted} #2}\\[0.15em]
  #4
}
