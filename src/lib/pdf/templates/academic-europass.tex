% Translate the renderer's English section names to Europass conventions.
\newcommand{\resumesection}[1]{%
  \def\tmptitle{#1}%
  \IfStrEq{#1}{Summary}{\def\tmptitle{Personal Profile}}{}%
  \IfStrEq{#1}{Experience}{\def\tmptitle{Work \& Research Experience}}{}%
  \IfStrEq{#1}{Projects}{\def\tmptitle{Research Projects}}{}%
  \IfStrEq{#1}{Education}{\def\tmptitle{Education \& Training}}{}%
  \IfStrEq{#1}{Skills}{\def\tmptitle{Digital \& Analytical Skills}}{}%
  \IfStrEq{#1}{Certifications}{\def\tmptitle{Publications \& Awards}}{}%
  \IfStrEq{#1}{Languages}{\def\tmptitle{Languages}}{}%
  \vspace{0.95em}%
  {\color{accent}\bfseries\large \tmptitle}\par
  \vspace{0.1em}{\textcolor{accent}{\rule{\textwidth}{0.6pt}}}\par
  \vspace{0.4em}%
}

\newcommand{\resumeentry}[4]{%
  {\bfseries\color{ink} #1}\hfill{\color{accent}#3}\\[0.05em]
  {\itshape\color{muted} #2}\\[0.12em]
  #4
}
