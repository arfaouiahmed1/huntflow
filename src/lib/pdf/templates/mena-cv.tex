% Translate the renderer's English section names to bilingual FR/EN headers.
\newcommand{\resumesection}[1]{%
  \def\tmptitle{#1}%
  \IfStrEq{#1}{Summary}{\def\tmptitle{Profil Professionnel / Professional Summary}}{}%
  \IfStrEq{#1}{Experience}{\def\tmptitle{Exp\'eriences / Experience}}{}%
  \IfStrEq{#1}{Projects}{\def\tmptitle{Projets / Projects}}{}%
  \IfStrEq{#1}{Education}{\def\tmptitle{Formation / Education}}{}%
  \IfStrEq{#1}{Skills}{\def\tmptitle{Comp\'etences / Skills}}{}%
  \IfStrEq{#1}{Certifications}{\def\tmptitle{Certifications}}{}%
  \IfStrEq{#1}{Languages}{\def\tmptitle{Langues / Languages}}{}%
  \vspace{0.9em}%
  {\color{accent}\bfseries\large \tmptitle}\par
  \vspace{0.1em}{\textcolor{accent}{\rule{\textwidth}{0.6pt}}}\par
  \vspace{0.4em}%
}

\newcommand{\resumeentry}[4]{%
  {\bfseries\color{ink} #1}\hfill{\color{accent}#3}\\[0.05em]
  {\itshape\color{muted} #2}\\[0.12em]
  #4
}
