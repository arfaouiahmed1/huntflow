% Translate the renderer's English section names to German.
\newcommand{\resumesection}[1]{%
  \def\tmptitle{#1}%
  \IfStrEq{#1}{Summary}{\def\tmptitle{Zusammenfassung}}{}%
  \IfStrEq{#1}{Experience}{\def\tmptitle{Berufserfahrung}}{}%
  \IfStrEq{#1}{Projects}{\def\tmptitle{Projekte}}{}%
  \IfStrEq{#1}{Education}{\def\tmptitle{Ausbildung}}{}%
  \IfStrEq{#1}{Skills}{\def\tmptitle{Kenntnisse}}{}%
  \IfStrEq{#1}{Certifications}{\def\tmptitle{Zertifikate}}{}%
  \IfStrEq{#1}{Languages}{\def\tmptitle{Sprachen}}{}%
  \vspace{0.9em}%
  {\color{accent}\bfseries\large \tmptitle}\par
  \vspace{0.1em}{\textcolor{accent}{\rule{\textwidth}{0.6pt}}}\par
  \vspace{0.4em}%
}

\newcommand{\resumeentry}[4]{%
  {\bfseries\color{ink} #1}\hfill{\color{accent}#3}\\[0.05em]
  {\itshape\color{muted} #2}\\[0.12em]
  #4
}
